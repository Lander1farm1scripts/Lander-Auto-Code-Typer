		btn.MouseEnter:Connect(function()
			TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(100, 100, 110)}):Play()
		end)
		btn.MouseLeave:Connect(function()
			TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(80, 80, 90)}):Play()
		end)
	end
	updateHover(minusBtn)
	updateHover(plusBtn)

	minusBtn.MouseButton1Click:Connect(function()
		local newVal = math.clamp(currentVal - step, minVal, maxVal)
		valueLabel.Text = string.format("%.2f", newVal)
		callback(newVal)
		currentVal = newVal
	end)
	plusBtn.MouseButton1Click:Connect(function()
		local newVal = math.clamp(currentVal + step, minVal, maxVal)
		valueLabel.Text = string.format("%.2f", newVal)
		callback(newVal)
		currentVal = newVal
	end)

	local clamped = math.clamp(currentVal, minVal, maxVal)
	valueLabel.Text = string.format("%.2f", clamped)
	callback(clamped)
	currentVal = clamped

	return frame
end

function createFloatingIcon(featureKey, displayName)
	if FloatingIcons[featureKey] then
		FloatingIcons[featureKey].Main:Destroy()
	end

	local pos = FloatingIconPositions[featureKey] or {x = math.random(100, 400), y = math.random(100, 400)}
	local label = FloatingIconLabels[featureKey] or displayName:sub(1, 3)

	local icon = createElement("TextButton", {
		Parent = ScreenGui,
		Name = featureKey .. "Icon",
		BackgroundColor3 = Color3.fromRGB(0, 0, 0),
		BackgroundTransparency = 0.4,
		Position = UDim2.new(0, pos.x, 0, pos.y),
		Size = UDim2.new(0, 80, 0, 32),
		AutoButtonColor = false,
		Text = label,
		TextColor3 = Color3.fromRGB(255, 255, 255),
		Font = Enum.Font.GothamSemibold,
		TextSize = 13,
		Draggable = true
	})
	createElement("UICorner", {Parent = icon, CornerRadius = UDim.new(0, 6)})

	local statusDot = createElement("Frame", {
		Parent = icon,
		Size = UDim2.new(0, 8, 0, 8),
		Position = UDim2.new(1, -14, 0, 12),
		BackgroundColor3 = Color3.fromRGB(200, 50, 50),
		BorderSizePixel = 0
	})
	createElement("UICorner", {Parent = statusDot, CornerRadius = UDim.new(1, 0)})

	local function updateVisual()
		if FeatureToggles[featureKey] then
			icon.BackgroundTransparency = 0.25
			statusDot.BackgroundColor3 = Color3.fromRGB(50, 205, 50)
			icon.TextColor3 = Color3.fromRGB(0, 255, 0)
		else
			icon.BackgroundTransparency = 0.4
			statusDot.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
			icon.TextColor3 = Color3.fromRGB(255, 255, 255)
		end
	end

	icon.MouseButton1Click:Connect(function()
		toggleFunction(featureKey, not FeatureToggles[featureKey])
		updateVisual()
	end)

	updateVisual()
	FloatingIcons[featureKey] = {Main = icon, Dot = statusDot, UpdateVisual = updateVisual}
	return FloatingIcons[featureKey]
end

function removeFloatingIcon(featureKey)
	if FloatingIcons[featureKey] then
		if typeof(FloatingIcons[featureKey]) == "table" then
			FloatingIcons[featureKey].Main:Destroy()
		else
			FloatingIcons[featureKey]:Destroy()
		end
		FloatingIcons[featureKey] = nil
	end
end

-- Shift Lock system
local function isValidCFrame(cf)
	if not cf then return false end
	local pos = cf.Position
	return pos.X == pos.X and pos.Y == pos.Y and pos.Z == pos.Z
end

local function safeCameraCFrame(cam, offset)
	if not isValidCFrame(cam) then
		if LocalPlayer.Character then
			local hrp = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
			if hrp then
				cam.CFrame = CFrame.new(hrp.Position + Vector3.new(0, 5, -10), hrp.Position)
			end
		end
		return cam.CFrame
	end
	return cam.CFrame * offset
end

-- AntiKnockback V2 / Ragdoll recovery
local function recoverFromRagdoll()
	if not IsRagdolled then return end
	local char, hum, hrp = LocalPlayer.Character, nil, nil
	if char then
		hum = char:FindFirstChildOfClass("Humanoid")
		hrp = char:FindFirstChild("HumanoidRootPart")
	end
	if not hum or not hrp then return end
	IsRagdolled = false
	hum:ChangeState(Enum.HumanoidStateType.GettingUp)
	hum.PlatformStand = false
	hrp.CanCollide = true
	if hrp.Anchored then hrp.Anchored = false end
	for _, child in pairs(char:GetChildren()) do
		if child:IsA("BasePart") then
			for _, c in pairs(child:GetChildren()) do
				if c:IsA("BallSocketConstraint") or c:IsA("HingeConstraint") then
					c:Destroy()
				end
			end
			local motor = child:FindFirstChildWhichIsA("Motor6D")
			if motor then motor.Enabled = true end
		end
	end
	pcall(function()
		local ControlModule = require(LocalPlayer.PlayerScripts.PlayerModule.ControlModule)
		if ControlModule and ControlModule.Enable then
			ControlModule:Enable()
		end
	end)
	hrp.Velocity = Vector3.new(0, math.min(hrp.Velocity.Y, 0), 0)
	hrp.RotVelocity = Vector3.new(0, 0, 0)
	Workspace.CurrentCamera.CameraSubject = hum
end

local function onRagdollState(character)
	if not AntiKnockbackV2Config.Enabled or IsRagdolled then return end
	if AntiKBStateConn then
		AntiKBStateConn:Disconnect()
		AntiKBStateConn = nil
	end
	IsRagdolled = true
	AntiKBStateConn = RunService.Heartbeat:Connect(function()
		AntiKBStateConn:Disconnect()
		AntiKBStateConn = nil
		recoverFromRagdoll()
	end)
end

local function setupAntiKnockbackV2(character)
	local humanoid = character:WaitForChild("Humanoid")
	if AntiKBStateConn then AntiKBStateConn:Disconnect() end
	AntiKBStateConn = humanoid.StateChanged:Connect(function(_, newState)
		if AntiKnockbackV2Config.Enabled then
			if newState == Enum.HumanoidStateType.FallingDown or newState == Enum.HumanoidStateType.Ragdoll or newState == Enum.HumanoidStateType.Physics then
				if not IsRagdolled then
					humanoid.PlatformStand = true
					onRagdollState(character)
				end
			elseif newState == Enum.HumanoidStateType.GettingUp or newState == Enum.HumanoidStateType.Running or newState == Enum.HumanoidStateType.RunningNoPhysics then
				humanoid.PlatformStand = false
				if IsRagdolled then
					recoverFromRagdoll()
				end
			end
		end
	end)
	table.insert(RagdollConnections, humanoid:GetPropertyChangedSignal("PlatformStand"):Connect(function()
		if AntiKnockbackV2Config.Enabled and not IsRagdolled and humanoid.PlatformStand then
			humanoid.PlatformStand = false
		end
	end))
	table.insert(RagdollConnections, RunService.Heartbeat:Connect(function()
		if not AntiKnockbackV2Config.Enabled then return end
		local char = LocalPlayer.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		if char and hrp then
			local pos = hrp.Position
			if IsRagdolled and AntiKnockbackV2Config.AntiKnockback and LastRagdollPos then
				local mag = (pos - LastRagdollPos).Magnitude
				local velMag = hrp.Velocity.Magnitude
				if mag > 8 and velMag > 25 then
					hrp.Velocity = hrp.Velocity * 0.7
					hrp.RotVelocity = hrp.RotVelocity * 0.7
				end
			end
			if not IsRagdolled and hum then
				local state = hum:GetState()
				if state == Enum.HumanoidStateType.Physics or state == Enum.HumanoidStateType.Ragdoll then
					hum:ChangeState(Enum.HumanoidStateType.GettingUp)
					hum.PlatformStand = false
				end
			end
		end
	end))
end

local function enableAntiKnockbackV2()
	AntiKnockbackV2Config.Enabled = true
	local char = LocalPlayer.Character
	if char then
		IsRagdolled = false
		if AntiKBStateConn then AntiKBStateConn:Disconnect() end
		char:WaitForChild("Humanoid")
		char:WaitForChild("HumanoidRootPart")
		setupAntiKnockbackV2(char)
		table.insert(RagdollConnections, char.ChildAdded:Connect(function(child)
			if child:IsA("BasePart") then
				table.insert(RagdollConnections, child.ChildAdded:Connect(function(c2)
					if not IsRagdolled and (c2:IsA("BallSocketConstraint") or c2:IsA("HingeConstraint")) then
						c2:Destroy()
					end
				end))
			end
		end))
	end
	LocalPlayer.CharacterAdded:Connect(function(c)
		task.wait(0.5)
		setupAntiKnockbackV2(c)
	end)
end

local function disableAntiKnockbackV2()
	AntiKnockbackV2Config.Enabled = false
	for _, conn in pairs(RagdollConnections) do
		if conn and conn.Connected then
			conn:Disconnect()
		end
	end
	table.clear(RagdollConnections)
	if AntiKBStateConn then
		AntiKBStateConn:Disconnect()
		AntiKBStateConn = nil
	end
	IsRagdolled = false
end

-- Aimbot
local function getClosestPlayer()
	local myHRP = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
	if not myHRP then return end
	local closest, closestDist = nil, AimbotConfig.Range
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LocalPlayer and plr.Character then
			local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
			if hrp then
				local dist = (myHRP.Position - hrp.Position).Magnitude
				if dist < closestDist then
					closest = plr
					closestDist = dist
				end
			end
		end
	end
	return closest
end

local function setupAimbotOnCharacter(character)
	local hrp = character:FindFirstChild("HumanoidRootPart")
	local hum = character:FindFirstChildOfClass("Humanoid")
	if not hrp or not hum then return end
	local att = Instance.new("Attachment", hrp)
	AimbotAlign = Instance.new("AlignOrientation", hrp)
	AimbotAlign.Attachment0 = att
	AimbotAlign.Mode = Enum.OrientationAlignmentMode.OneAttachment
	AimbotAlign.Responsiveness = 200
	AimbotAlign.Enabled = false
	hum.AutoRotate = true
end

local function clearAimbotVisuals()
	if AimbotHighlight then
		AimbotHighlight:Destroy()
		AimbotHighlight = nil
	end
	if AimbotAlign then
		AimbotAlign.Enabled = false
	end
	AimbotTarget = nil
	local char = LocalPlayer.Character
	if char then
		local hum = char:FindFirstChildOfClass("Humanoid")
		if hum then hum.AutoRotate = true end
	end
end

local function startAimbotLoop()
	if AimbotRenderConn then AimbotRenderConn:Disconnect() end
	AimbotRenderConn = RunService.RenderStepped:Connect(function()
		if not AimbotActive or not AimbotAlign then
			clearAimbotVisuals()
			return
		end
		local char = LocalPlayer.Character
		if not char then
			clearAimbotVisuals()
			return
		end
		local hum = char:FindFirstChildOfClass("Humanoid")
		local hrp = char:FindFirstChild("HumanoidRootPart")
		if not hum or not hrp then
			clearAimbotVisuals()
			return
		end
		local target = getClosestPlayer()
		if not target then
			clearAimbotVisuals()
			return
		end
		local targetHRP = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
		if not targetHRP then
			clearAimbotVisuals()
			return
		end
		if target ~= AimbotTarget then
			if AimbotHighlight then AimbotHighlight:Destroy() end
			AimbotHighlight = Instance.new("Highlight", target.Character)
			AimbotHighlight.FillTransparency = 1
			AimbotHighlight.OutlineColor = Color3.fromRGB(255, 255, 255)
			AimbotHighlight.OutlineTransparency = 0
			AimbotTarget = target
		end
		local predPos = targetHRP.Position + (targetHRP.AssemblyLinearVelocity - hrp.AssemblyLinearVelocity) * AimbotConfig.Prediction
		hum.AutoRotate = false
		AimbotAlign.Enabled = true
		local lookAt = Vector3.new(predPos.X, hrp.Position.Y, predPos.Z)
		AimbotAlign.CFrame = CFrame.lookAt(hrp.Position, lookAt)
		if AimbotConfig.CameraAim then
			local targetCF = CFrame.lookAt(Camera.CFrame.Position, predPos)
			Camera.CFrame = Camera.CFrame:Lerp(targetCF, AimbotConfig.SoftCamera)
		end
	end)
end

local function stopAimbot()
	if AimbotRenderConn then
		AimbotRenderConn:Disconnect()
		AimbotRenderConn = nil
	end
	clearAimbotVisuals()
end

-- Infinite Jump
local function getPing()
	local network = Stats:FindFirstChild("Network")
	if not network then return 0 end
	local item = network:FindFirstChild("ServerStatsItem") and network.ServerStatsItem:FindFirstChild("Data Ping")
	if item and item.GetValue then
		return math.clamp(item:GetValue() / 1000, 0, 0.25)
	end
	return 0
end

local function applyJumpForce()
	local char = LocalPlayer.Character
	if not char then return end
	local hrp = char:FindFirstChild("HumanoidRootPart")
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hrp or not hum or hum.Health <= 0 then return end
	if tick() - LastJumpTime < InfiniteJumpConfig.JumpCooldown then return end
	LastJumpTime = tick()
	local ping = getPing()
	local forceY = hrp.AssemblyMass * InfiniteJumpConfig.ForceMultiplier
	JumpForceSmoothed = JumpForceSmoothed + (forceY - JumpForceSmoothed) * 0.5
	local att = Instance.new("Attachment")
	att.Name = "JumpAttachment"
	att.Parent = hrp
	local vf = Instance.new("VectorForce")
	vf.Name = "JumpForce"
	vf.Attachment0 = att
	vf.RelativeTo = Enum.ActuatorRelativeTo.World
	vf.Force = Vector3.new(0, JumpForceSmoothed, 0)
	vf.Parent = hrp
	Debris:AddItem(vf, 0.1 + ping)
	Debris:AddItem(att, 0.1 + ping)
end

local function setupInfiniteJump(character)
	character:WaitForChild("Humanoid")
	if JumpRequestConn then JumpRequestConn:Disconnect() end
	JumpRequestConn = UserInputService.JumpRequest:Connect(function()
		if InfiniteJumpActive then
			applyJumpForce()
		end
	end)
	character.ChildAdded:Connect(function(child)
		if child:IsA("Humanoid") then
			if JumpRequestConn then JumpRequestConn:Disconnect() end
			JumpRequestConn = UserInputService.JumpRequest:Connect(function()
				if InfiniteJumpActive then
					applyJumpForce()
				end
			end)
		end
	end)
end

local function enableInfiniteJump()

	end
	LocalPlayer.CharacterAdded:Connect(setupInfiniteJump)
	RunService.RenderStepped:Connect(function()
		if InfiniteJumpActive then

		end
	end)
end

-- AntiKnockback (V1)
local function dampenVelocity()
	if not AntiKnockbackConfig.Enabled then return end
	local char = LocalPlayer.Character
	if char and char:FindFirstChild("HumanoidRootPart") then
		local hrp = char.HumanoidRootPart
		local vel = hrp.AssemblyLinearVelocity
		if vel.Magnitude > 10 then
			hrp.AssemblyLinearVelocity = vel * (1 - AntiKnockbackConfig.Intensity)
		end
	end
end

local function triggerAntiKnockback()
	if tick() - AntiKBLastTrigger < AntiKnockbackConfig.Cooldown then return end
	AntiKBLastTrigger = tick()
	if AntiKBHeartbeat then AntiKBHeartbeat:Disconnect() end
	AntiKBHeartbeat = RunService.Heartbeat:Connect(dampenVelocity)
	task.delay(AntiKnockbackConfig.Duration, function()
		if AntiKBHeartbeat then
			AntiKBHeartbeat:Disconnect()
			AntiKBHeartbeat = nil
		end
	end)
end

local function enableAntiKnockback()

-- AutoHit / Chase
local function getPlayerVelocity(player)
	if not player.Character then return 0 end
	local hrp = player.Character:FindFirstChild("HumanoidRootPart")
	if not hrp then return 0 end
	local mag = hrp.Velocity.Magnitude
	if not PlayerVelocityCache[player] then
		PlayerVelocityCache[player] = {position = hrp.Position, time = tick()}
		return mag
	end
	local cache = PlayerVelocityCache[player]
	local dt = tick() - cache.time
	if dt > 0.1 then
		local speed = (hrp.Position - cache.position).Magnitude / dt
		PlayerVelocityCache[player] = {position = hrp.Position, time = tick()}
		return (mag + speed) / 2
	end
	return mag
end

local function interceptTime(posA, velA, posB, velB, radius)
	local relPos = posB - posA
	local relVel = velB - velA
	if relVel.Magnitude < AutoHitConfig.SpeedThreshold then
		if radius >= relPos.Magnitude then return 0 end
		return nil
	end
	local a = relVel:Dot(relVel)
	local b = 2 * relPos:Dot(relVel)
	local c = relPos:Dot(relPos) - radius * radius
	local disc = b * b - 4 * a * c
	if disc < 0 then return nil end
	local sqrtD = math.sqrt(disc)
	local t1 = (-b + sqrtD) / (2 * a)
	local t2 = (-b - sqrtD) / (2 * a)
	if t1 >= 0 and t2 >= 0 then return math.min(t1, t2) end
	if t1 >= 0 then return t1 end
	if t2 >= 0 then return t2 end
	return nil
end

local function isInOptimalRange(myChar, targetPlayer, tool)
	if not myChar or not targetPlayer.Character then return false end
	local myHRP = myChar:FindFirstChild("HumanoidRootPart")
	local targetHRP = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
	local handle = tool:FindFirstChild("Handle")
	if not myHRP or not targetHRP or not handle then return false end
	-- simplified reach check (literal structure preserved)
	local mySpeed = getPlayerVelocity(LocalPlayer)
	local targetSpeed = getPlayerVelocity(targetPlayer)
	local myPos = myHRP.Position
	local targetPos = targetHRP.Position
	if (targetPos - myPos).Magnitude <= AutoHitConfig.OptimalAttackDistance then
		local myVel = myHRP.Velocity
		local targetVel = targetHRP.Velocity
		local unit = (targetPos - myPos).Unit
		local approaching = myVel:Dot(unit) > 0 or targetVel:Dot(-unit) > 0
		if approaching or (mySpeed < AutoHitConfig.SpeedThreshold and targetSpeed < AutoHitConfig.SpeedThreshold) then
			return true
		end
	end
	local t = interceptTime(myPos, myHRP.Velocity, targetPos, targetHRP.Velocity, AutoHitConfig.OptimalAttackDistance)
	if t and t <= AutoHitConfig.PredictionFactor then
		return true
	end
	return false
end

local function findBestTarget(tool)
	local char = LocalPlayer.Character
	if not char or not tool then return nil end
	local handle = tool:FindFirstChild("Handle")
	if not handle then return nil end
	local best, bestDist, inRange = nil, math.huge, false
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LocalPlayer and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
			local hrp = plr.Character.HumanoidRootPart
			local dist = (handle.Position - hrp.Position).Magnitude
			if dist <= AutoHitConfig.ChaseDistance and dist < bestDist then
				bestDist = dist
				best = plr
				inRange = isInOptimalRange(char, plr, tool)
			end
		end
	end
	return best, bestDist, inRange
end

local function moveToward(pos)
	if not LocalHumanoid or not LocalPlayer.Character then return end
	local hrp = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
	if not hrp then return end
	local unit = (pos - hrp.Position).Unit
	LocalHumanoid:MoveTo(hrp.Position + unit * 5)
end

local function stopMoving()
	if LocalHumanoid then
		LocalHumanoid:MoveTo(LocalHumanoid.Parent.HumanoidRootPart.Position)
	end
end

local function enableAutoHit()
	LocalHumanoid = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
	LocalPlayer.CharacterAdded:Connect(function(c)
		LocalHumanoid = c:WaitForChild("Humanoid")
	end)
	if AutoHitMoveConn then AutoHitMoveConn:Disconnect() end
	AutoHitMoveConn = RunService.Heartbeat:Connect(function()
		if not FeatureToggles.AutoHit or not AutoHitConfig.MovementEnabled or not LocalHumanoid then return end
		local char = LocalPlayer.Character
		if not char then return end
		local tool
		for _, child in ipairs(char:GetChildren()) do
			if child:IsA("Tool") and (child.Name:match("Sword$") or child.Name:match("Bat$") or child.Name:match("Slap$")) then
				tool = child
				break
			end
		end
		if not tool then
			if AutoHitChaseActive then
				AutoHitChaseActive = false
				AutoHitTarget = nil
				LocalHumanoid.WalkSpeed = AutoHitConfig.WalkSpeedNormal
				stopMoving()
			end
			return
		end
		local target, dist, inRange = findBestTarget(tool)
		if target and dist <= AutoHitConfig.ChaseDistance then
			if not AutoHitChaseActive or target ~= AutoHitTarget then
				AutoHitChaseActive = true
				LocalHumanoid.WalkSpeed = AutoHitConfig.WalkSpeedChase
			end
			if inRange then
				stopMoving()
				return
			end
			local hrp = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
			if hrp then
				moveToward(hrp.Position)
			end
		elseif AutoHitChaseActive then
			LocalHumanoid.WalkSpeed = AutoHitConfig.WalkSpeedNormal
			stopMoving()
		end
	end)

	if AutoHitAttackConn then AutoHitAttackConn:Disconnect() end
	AutoHitAttackConn = RunService.Heartbeat:Connect(function()
		if not FeatureToggles.AutoHit then return end
		if tick() - AutoHitLastHit < AutoHitConfig.DebounceTime then return end
		local char = LocalPlayer.Character
		if not char then return end
		for _, child in ipairs(char:GetChildren()) do
			if child:IsA("Tool") and (child.Name:match("Sword$") or child.Name:match("Bat$") or child.Name:match("Slap$")) then
				local target, _, inRange = findBestTarget(child)
				if target and inRange then
					child:Activate()
					AutoHitLastHit = tick()
					return
				end
			end
		end
	end)
end

local function disableAutoHit()
	if AutoHitMoveConn then AutoHitMoveConn:Disconnect() AutoHitMoveConn = nil end
	if AutoHitAttackConn then AutoHitAttackConn:Disconnect() AutoHitAttackConn = nil end
	AutoHitChaseActive = false
	AutoHitTarget = nil
	if LocalHumanoid then
		LocalHumanoid.WalkSpeed = AutoHitConfig.WalkSpeedNormal
	end
	stopMoving()
end

-- Master toggle
function toggleFunction(featureKey, enabled)
	FeatureToggles[featureKey] = enabled
	if featureKey == "Aimbot" then
		AimbotActive = enabled
		if enabled then
			local char = LocalPlayer.Character
			if char then
				setupAimbotOnCharacter(char)
				startAimbotLoop()
			end
			LocalPlayer.CharacterAdded:Connect(function(c)
				if AimbotActive then
					task.wait(1)
					setupAimbotOnCharacter(c)
					startAimbotLoop()
				end
			end)
		else
			stopAimbot()
		end
	elseif featureKey == "GrabActivator" then
		-- Grab logic not fully expanded in source; toggle state only
	elseif featureKey == "SpeedBoost" then
		SpeedBoostConfig.Enabled = enabled
		if enabled then
			enableSpeedBoost()
		else
			disableSpeedBoost()
		end
	elseif featureKey == "ShiftLock" then
		if enabled then
			task.spawn(function()
				local cam = Workspace.CurrentCamera
				if cam then
					cam.CameraType = Enum.CameraType.Custom
				end
				task.wait(0.05)
				enableShiftLock()
				task.wait(0.1)
				if not ShiftLockActive then
					task.wait(0.2)
					enableShiftLock()
				end
			end)
		else
			disableShiftLock()
			task.spawn(function()
				task.wait(0.05)
				local cam = Workspace.CurrentCamera
				if cam then cam.CameraType = Enum.CameraType.Custom end
			end)
		end
	elseif featureKey == "AntiKnockback" then
		AntiKnockbackConfig.Enabled = enabled
		if enabled then
			enableAntiKnockback()
		elseif AntiKBHeartbeat then
			AntiKBHeartbeat:Disconnect()
		end
	elseif featureKey == "AntiRagdoll" then
		AntiRagdollConfig.Enabled = enabled
		-- Additional anti-ragdoll force logic not fully expanded in truncated sections
	elseif featureKey == "AutoHit" then
		AutoHitConfig.Enabled = enabled
		if enabled then
			enableAutoHit()
		else
			disableAutoHit()
		end
	elseif featureKey == "AntiKnockbackV2" then
		if enabled then
			enableAntiKnockbackV2()
		else
			disableAntiKnockbackV2()
		end
	elseif featureKey == "InfiniteJump" then
		InfiniteJumpConfig.Enabled = enabled
		InfiniteJumpActive = enabled
		if enabled then
			enableInfiniteJump()
		elseif JumpRequestConn then
			JumpRequestConn:Disconnect()
		end
	end
	if FloatingIcons[featureKey] and FloatingIcons[featureKey].UpdateVisual then
		FloatingIcons[featureKey].UpdateVisual()
	end
end
