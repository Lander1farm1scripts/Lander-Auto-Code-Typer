local function corner(obj, r)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, r or 8)
    c.Parent = obj
    return c
end

local function stroke(obj, col, t, tr)
    local s = Instance.new("UIStroke")
    s.Color = col or COLORS.border
    s.Thickness = t or 1
    s.Transparency = tr or 0.3
    s.Parent = obj
    return s
end

local function padding(obj, l, r, t, b)
    local p = Instance.new("UIPadding")
    p.PaddingLeft = UDim.new(0, l or 0)
    p.PaddingRight = UDim.new(0, r or 0)
    p.PaddingTop = UDim.new(0, t or 0)
    p.PaddingBottom = UDim.new(0, b or 0)
    p.Parent = obj
    return p
end

local function createButton(parent, text, size, pos, bgColor, textColor, fontSize)
    local btn = Instance.new("TextButton")
    btn.Size = size
    btn.Position = pos
    btn.BackgroundColor3 = bgColor or COLORS.panel3
    btn.BorderSizePixel = 0
    btn.Text = text
    btn.TextColor3 = textColor or COLORS.text
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = fontSize or 12
    btn.AutoButtonColor = false
    btn.Parent = parent
    corner(btn, 8)
    stroke(btn, COLORS.border, 1.5, 0.4)
    local orig = bgColor or COLORS.panel3
    btn.MouseEnter:Connect(function() TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = orig:Lerp(Color3.new(1,1,1), 0.1)}):Play() end)
    btn.MouseLeave:Connect(function() TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = orig}):Play() end)
    btn.MouseButton1Down:Connect(function() TweenService:Create(btn, TweenInfo.new(0.05), {BackgroundColor3 = orig:Lerp(Color3.new(0,0,0), 0.2)}):Play() end)
    btn.MouseButton1Up:Connect(function() TweenService:Create(btn, TweenInfo.new(0.05), {BackgroundColor3 = orig}):Play() end)
    return btn
end

local function loadConfig()
    if not (readfile and isfile and isfile(CONFIG_FILE)) then return end
    local ok, decoded = pcall(function() return HttpService:JSONDecode(readfile(CONFIG_FILE)) end)
    if not ok or type(decoded) ~= "table" then return end
    for key, value in pairs(decoded) do 
        if config[key] ~= nil then config[key] = value end 
    end
end

local function saveConfig()
    if not (writefile and readfile) then return end
    pcall(function() writefile(CONFIG_FILE, HttpService:JSONEncode(config)) end)
end

            local codesRoot = PlayerGui:FindFirstChild("Codes")
            if not codesRoot then return end
            local codes = codesRoot:FindFirstChild("Codes")
            if not codes then return end
            local redeem = codes:FindFirstChild("CodeRedeem")
            if not redeem then return end
            local textBox = redeem:FindFirstChild("TextBox")
            if not textBox then return end
            local confirm = codes:FindFirstChild("Confirm")
            if not confirm then return end
            
            textBox.Text = code
            task.wait(0.05)
            local button = confirm:FindFirstChildWhichIsA("TextButton") or (confirm:IsA("TextButton") and confirm or nil)
            if button and button:IsA("GuiButton") then
                button:Activate()
                success = true
            end
        end)
        if success then
            lastCode = code
            sessionStats.codesRedeemed = sessionStats.codesRedeemed + 1
            logStatus("✓ Redeemed: " .. code, "success")
        else
            logStatus("✗ UI unavailable for: " .. code, "error")
        end
        updatePreview()
        return success
    end

    local function startPulse()
        if pulseTween then pulseTween:Cancel() pulseTween = nil end
        pulseTween = TweenService:Create(force, TweenInfo.new(0.65, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true), {BackgroundTransparency = 0.25})
        pulseTween:Play()
    end

    local function cleanAndValidateText(text)
        if type(text) ~= "string" then return nil end
        text = text:gsub("<.->", ""):gsub("%s+", " "):gsub("^%s+", ""):gsub("%s+$", "")
        return text ~= "" and text or nil
    end

    local function matchesKeyword(text)
        local lower = string.lower(text or "")
        for _, keyword in ipairs(keywords or {}) do
            if keyword and keyword ~= "" and string.find(lower, string.lower(keyword), 1, true) then return true end
        end
        return false
    end

    local function applyReplacement(text)
        local lower = string.lower(text)
        for _, rule in ipairs(replacements or {}) do
            local kw = tostring(rule.kw or "")
            local rep = tostring(rule.rep or "")
            if kw ~= "" and string.find(lower, string.lower(kw), 1, true) then return rep end
        end
        return nil
    end

    local function parseRewardFromText(rawText)
        if not rawText then return nil end
        local text = rawText:gsub("<.->", "")
        local lower = string.lower(text)
        local patterns = {
            {match = "code redeemed for ", after = true},
            {match = "code redeemed: ", after = true},
            {match = "you got ", after = true},
            {match = "you received ", after = true},
            {match = " spawned", after = false},
        }
        for _, p in ipairs(patterns) do
            local startIdx, endIdx = string.find(lower, p.match, 1, true)
            if startIdx then
                local name = nil
                if p.after then name = string.sub(text, endIdx + 1) else name = string.sub(text, 1, startIdx - 1) end
                name = name:gsub("^%s+", ""):gsub("%s+$", ""):gsub("[!%.]+$", "")
                if name ~= "" then return name end
            end
        end
        return nil
    end

    local function setForceScan(state)
        if state then
            local n = captureCount > 0 and captureCount or 1
            collecting = true
            forceScan = true
            collected = {}
            remaining = n
            collectDeadline = os.clock() + 60
            startPulse()
            updatePreview()
            force.Text = "✖ CANCEL SCAN"
            force.BackgroundColor3 = COLORS.bad
            logStatus("Force scan armed — waiting for " .. n .. " notification(s)", "action")
        else
            resetCollection()
            force.Text = "⚡ FORCE SCAN + REDEEM"
            force.BackgroundColor3 = COLORS.accent
        end
    end

    local function sendSnipeWebhook(rewardName)
        local requestFunc = (syn and syn.request) or (http and http.request) or (http_request) or request
        if not requestFunc then return end
        local playerCount = #Players:GetPlayers()
        local isPrivate = (game.PrivateServerId ~= "" and game.PrivateServerId ~= "0")
        local serverType = isPrivate and "🔒 Private" or "🌍 Public"
        local profileLink = "https://www.roblox.com/users/" .. player.UserId .. "/profile"
        local codeUsed = (lastCode ~= "" and lastCode) or "N/A"
        local fields = {
            {name = "👤 User", value = player.DisplayName .. " (@" .. player.Name .. ")", inline = true},
            {name = "🎟️ Code", value = "`" .. codeUsed .. "`", inline = true},
            {name = "👥 Players", value = tostring(playerCount), inline = true},
            {name = "🌐 Server", value = serverType, inline = true},
            {name = "🔗 Profile", value = "[Click to view](" .. profileLink .. ")", inline = false}
        }
        local payload = {
            embeds = {
                {
                    title = "🌮 Brainrot Sniped!",
                    description = "**" .. rewardName .. "** just entered the base!",
                    color = 10181046, 
                    fields = fields,
                    footer = {text = "Lander Auto Typer"},
                    timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ")
                }
            }
        }
        task.spawn(function()
            pcall(function()
                requestFunc({
                    Url = DISCORD_WEBHOOK_URL, 
                    Method = "POST",
                    Headers = { ["Content-Type"] = "application/json" },
                    Body = HttpService:JSONEncode(payload)
                })
            end)
        end)
    end

    local function dispatch(rawText)
        if not rawText or rawText == "" then return end
        local rewardName = parseRewardFromText(rawText)
        if rewardName then
            sessionStats.brainrotsSniped = sessionStats.brainrotsSniped + 1
            local msg = "🌮 Sniped " .. rewardName .. "!"
            logStatus(msg, "success")
            addSnipe(msg, COLORS.good)
            sendSnipeWebhook(rewardName)
            return
        end
        local text = cleanAndValidateText(rawText)
        if not text then return end
        local now = os.clock()
        if text == lastDispatchedText and (now - lastDispatchTime) < 0.2 then return end
        lastDispatchedText = text
        lastDispatchTime = now
        if collecting and collectDeadline > 0 and now > collectDeadline then
            logStatus("Collection timed out", "warn")
            if forceScan then setForceScan(false) else resetCollection() end
            return
        end
        if forceScan and collecting then
            table.insert(collected, text)
            remaining -= 1
            redeemDeadline = now + 0.1
            updatePreview()
            logStatus("Collected [" .. (#collected) .. "]: " .. text, "info")
            return
        end
        if captureCount > 0 then
            if collecting then
                local replacement = applyReplacement(text)
                local piece = replacement ~= nil and replacement or text
                table.insert(collected, piece)
                remaining -= 1
                redeemDeadline = now + 0.1
                updatePreview()
                logStatus("Collected [" .. (#collected) .. "]: " .. piece, "info")
            elseif matchesKeyword(rawText) then
                collecting = true
                collected = {}
                remaining = captureCount
                collectDeadline = now + 60
                redeemDeadline = now + 0.1
                startPulse()
                updatePreview()
                logStatus("Trigger found — collecting " .. captureCount .. " notification(s)", "action")
                local replacement = applyReplacement(text)
                local piece = replacement ~= nil and replacement or text
                table.insert(collected, piece)
                remaining -= 1
            end
            return
        end
        if matchesKeyword(rawText) then
            local replacement = applyReplacement(text)
            local result = replacement ~= nil and replacement or text
            if result ~= "" then
                lastCode = result
                updatePreview()
                logStatus("Instant code: " .. result, "action")
                if autoCode then
                    local capturedResult = result
                    task.delay(0.1, function()
                        if lastCode == capturedResult then redeemCode(capturedResult) end
                    end)
                end
            end
        end
    end

    task.spawn(function()
        while true do
            task.wait(0.05)
            if collecting and #collected > 0 and remaining <= 0 and redeemDeadline > 0 and os.clock() > redeemDeadline then
                local result = table.concat(collected)
                local wasForce = forceScan
                resetCollection()
                if result ~= "" then
                    if wasForce then
                        logStatus("Force scan assembled: " .. result, "action")
                        redeemCode(result)
                    elseif autoCode then
                        logStatus("Code assembled: " .. result, "action")
                        redeemCode(result)
                    end
                end
            end
        end
    end)

    local function ownedByUs(obj)
        local current = obj
        while current and current ~= game do
            if current == gui then return true end
            current = current.Parent
        end
        return false
    end

    local function hookLabel(label)
        if seen[label] or ownedByUs(label) then return end
        seen[label] = true
        label:GetPropertyChangedSignal("Text"):Connect(function()
            if ownedByUs(label) or label.Text == "" then return end
            dispatch(label.Text)
        end)
        if label.Text and label.Text ~= "" then dispatch(label.Text) end
    end
    
    local function hookContainer(container)
        if not container then return end
        container.DescendantAdded:Connect(function(child)
            if child:IsA("TextLabel") then hookLabel(child) end
        end)
    end

    local function hookNotificationUI()
        local notifNames = {"TopNotification", "Notification", "Notifications", "Notify", "Toast", "Alert", "MessageLabel"}
        for _, name in ipairs(notifNames) do
            local existing = PlayerGui:FindFirstChild(name)
            if existing then hookContainer(existing) end
        end
        PlayerGui.ChildAdded:Connect(function(child)
            for _, name in ipairs(notifNames) do
                if child.Name == name then hookContainer(child) break end
            end
        end)
    end
