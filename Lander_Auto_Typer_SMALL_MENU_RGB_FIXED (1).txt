-- FIX: All HTTP requests are wrapped in task.spawn to prevent game freezing
local function relayRequest(url, payload, callback)
    local requestFunc = (syn and syn.request) or (http and http.request) or (http_request) or request
    if not requestFunc then if callback then callback(nil, "HTTP unavailable") end return end
    task.spawn(function()
        local ok, response = pcall(function() return requestFunc({Url = url, Method = "POST", Headers = { ["Content-Type"] = "application/json" }, Body = HttpService:JSONEncode(payload)}) end)
        if not ok or not response then if callback then callback(nil, "Request failed") end return end
        local body = response.Body or response.body or "{}"
        local decodeOk, data = pcall(function() return HttpService:JSONDecode(body) end)
        if not decodeOk then if callback then callback(nil, "Invalid response") end return end
        local status = tonumber(response.StatusCode or 0) or 0
        if status < 200 or status >= 300 then if callback then callback(nil, data.error or "Invalid key") end return end
        if callback then callback(data, nil) end
    end)
end

local function loadConfig()
    if not (readfile and isfile and isfile(CONFIG_FILE)) then return end
    local ok, decoded = pcall(function() return HttpService:JSONDecode(readfile(CONFIG_FILE)) end)
    if not ok or type(decoded) ~= "table" then return end
    for key, value in pairs(decoded) do if config[key] ~= nil then config[key] = value end end
end
local function saveConfig()
    if not (writefile and readfile) then return end
    pcall(function() writefile(CONFIG_FILE, HttpService:JSONEncode(config)) end)
end

    local function cleanAndValidateText(text)
        if type(text) ~= "string" then return nil end
        text = text:gsub("<.->", ""):gsub("%s+", " "):gsub("^%s+", ""):gsub("%s+$", "")
        return text ~= "" and text or nil

    local function matchesKeyword(text)
        local lower = string.lower(text or "")
        for _, keyword in ipairs(keywords or {}) do
            if keyword and keyword ~= "" and string.find(lower, string.lower(keyword), 1, true) then return true end
        end
        return false
    end
    local function applyReplacement(text)
        local lower = string.lower(text)
        for _, rule in ipairs(replacements or {}) do
            local kw = tostring(rule.kw or "") local rep = tostring(rule.rep or "")
            if kw ~= "" and string.find(lower, string.lower(kw), 1, true) then return rep end
        end
        return nil
    end
    local function parseRewardFromText(rawText)
        if not rawText then return nil end
        local text = rawText:gsub("<.->", "") local lower = string.lower(text)
        local patterns = {
            {match = "code redeemed for ", after = true}, {match = "code redeemed: ", after = true},
            {match = "you got ", after = true}, {match = "you received ", after = true},
            {match = " spawned", after = false},
        }
        for _, p in ipairs(patterns) do
            local startIdx, endIdx = string.find(lower, p.match, 1, true)
            if startIdx then
                local name = nil
                if p.after then name = string.sub(text, endIdx + 1) else name = string.sub(text, 1, startIdx - 1) end
                name = name:gsub("^%s+", ""):gsub("%s+$", ""):gsub("[!%.]+$", "")
                if name ~= "" then return name end
            end
        end
        return nil
    end

    local function dispatch(rawText)
        if not rawText or rawText == "" then return end
        local rewardName = parseRewardFromText(rawText)
        if rewardName then
            sessionStats.brainrotsSniped = sessionStats.brainrotsSniped + 1
            local msg = "🌮 Sniped " .. rewardName .. "!" logStatus(msg, "success") addSnipe(msg, COLORS.good) sendSnipeWebhook(rewardName) return
        end
        local text = cleanAndValidateText(rawText) if not text then return end
        local now = os.clock()
        if text == lastDispatchedText and (now - lastDispatchTime) < 0.2 then return end
        lastDispatchedText = text lastDispatchTime = now
        if collecting and collectDeadline > 0 and now > collectDeadline then logStatus("Collection timed out", "warn") if forceScan then setForceScan(false) else resetCollection() end return end
        if forceScan and collecting then
            table.insert(collected, text) remaining -= 1 redeemDeadline = now + 0.1 updatePreview() logStatus("Collected [" .. (#collected) .. "]: " .. text, "info") return
        end
        if captureCount > 0 then
            if collecting then
                local replacement = applyReplacement(text) local piece = replacement ~= nil and replacement or text
                table.insert(collected, piece) remaining -= 1 redeemDeadline = now + 0.1 updatePreview() logStatus("Collected [" .. (#collected) .. "]: " .. piece, "info")
            elseif matchesKeyword(rawText) then
                collecting = true collected = {} remaining = captureCount collectDeadline = now + 60 redeemDeadline = now + 0.1 startPulse() updatePreview() logStatus("Trigger found — collecting " .. captureCount .. " notification(s)", "action")
                local replacement = applyReplacement(text) local piece = replacement ~= nil and replacement or text table.insert(collected, piece) remaining -= 1
            end
            return
        end
        if matchesKeyword(rawText) then
            local replacement = applyReplacement(text) local result = replacement ~= nil and replacement or text
            if result ~= "" then
                lastCode = result updatePreview() logStatus("Instant code: " .. result, "action")
                if autoCode then local capturedResult = result task.delay(0.1, function() if lastCode == capturedResult then redeemCode(capturedResult) end end) end
            end
        end
    end
